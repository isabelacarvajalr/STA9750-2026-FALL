get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

Calculating Total Emissions

Here we combine the three tables NTD_SERVICE, NTD_ENERGY, and EIA_SEP_REPORT together to be able to get the CO2 associated with each agency and mode.

Show code 
# NTD_SERVICE has several rows per agency + mode (split by type of service), so add
# them up first. Also convert the ID from text ("00001") to a number to match NTD_ENERGY.
service_by_mode <- NTD_SERVICE |>
  mutate(NTD_ID = as.numeric(NTD_ID)) |>
  group_by(NTD_ID, AGENCY, state, mode) |>
  summarize(
    total_upt       = sum(total_upt, na.rm = TRUE),
    passenger_miles = sum(sum_passenger_miles, na.rm = TRUE),
    .groups = "drop"
  )
Show code 
# Join 1: fuel use + ridership, matched on agency ID AND mode (inner: need both)
# Join 2: + each state's CO2 per MWh for electric vehicles (inner: 50 states only)
emissions <- NTD_ENERGY |>
  inner_join(service_by_mode, join_by(`NTD ID` == NTD_ID, Mode == mode)) |>
  inner_join(EIA_SEP_REPORT |> select(abbreviation, CO2_MWh),
             join_by(state == abbreviation))
Show code 
# EIA CO2 conversion factors (pounds of CO2 per gallon), from
# https://www.eia.gov/environment/emissions/co2_vol_mass.php
CO2_DIESEL   <- 22.45   # also used for bio-diesel
CO2_RESIDUAL <- 24.78   # bunker fuel
CO2_GASOLINE <- 18.73   # finished motor gasoline, also used for ethanol
CO2_KEROSENE <- 21.78
CO2_PROPANE  <- 12.68   # liquified petroleum gas
CO2_CNG      <- 120.85 * 0.12667   # natural gas per Mcf x Mcf per gallon equivalent
CO2_LNG      <- 120.85 * 0.0826    # natural gas per Mcf x Mcf per gallon of LNG

emissions <- emissions |>
  mutate(
    total_co2_lbs =
      (`Gallons of Diesel Fuel` + `Gallons of Bio-Diesel`) * CO2_DIESEL +
      `Gallons of Bunker Fuel`                             * CO2_RESIDUAL +
      (`Gallons of Gasoline` + `Gallons of Ethanol`)       * CO2_GASOLINE +
      `Gallons of Kerosene`                                * CO2_KEROSENE +
      `Gallons of Liquified Petroleum Gas`                 * CO2_PROPANE +
      `Gallons of Compressed Natural Gas`                  * CO2_CNG +
      `Gallons of Liquified Nat Gas`                       * CO2_LNG +
      # electricity: kWh -> MWh (divide by 1000) x the state's lbs CO2 per MWh
      (`Kilowatt hours of Electric Battery` +
         `Kilowatt hours of Electric Propulsion`) / 1000 * CO2_MWh,
    total_co2_tons = total_co2_lbs / 2000   # 2,000 lbs = 1 short ton
  )

I used inner_join to merge NTD_ENERGY with service_by_mode and then I merged that with EIA_SEP_REPORT. I joined them using inner_join because I needed ridership and fuel to compute the emissions. The Emissions were computed by multiplying each fuel with its EIA CO2 fuel factors data found in the EIA link EIA’s CO2 emissions coefficients.

Then the Electricity Emissions was computed by converting kWh to MWh (1MWh=1,000kWh) and then multiplying the MWh from NTD_ENERGY by the states CO2 per MWh that could be found in the EIA_SEP_REPORT. The fuel and electricity emissions are added for the total.

## Calculating Total Emissions

Here we combine the three tables NTD_SERVICE, NTD_ENERGY, and EIA_SEP_REPORT together to be able to get the CO2 associated with each agency and mode. 

```{r}
# NTD_SERVICE has several rows per agency + mode (split by type of service), so add
# them up first. Also convert the ID from text ("00001") to a number to match NTD_ENERGY.
service_by_mode <- NTD_SERVICE |>
  mutate(NTD_ID = as.numeric(NTD_ID)) |>
  group_by(NTD_ID, AGENCY, state, mode) |>
  summarize(
    total_upt       = sum(total_upt, na.rm = TRUE),
    passenger_miles = sum(sum_passenger_miles, na.rm = TRUE),
    .groups = "drop"
  )
```

```{r}
# Join 1: fuel use + ridership, matched on agency ID AND mode (inner: need both)
# Join 2: + each state's CO2 per MWh for electric vehicles (inner: 50 states only)
emissions <- NTD_ENERGY |>
  inner_join(service_by_mode, join_by(`NTD ID` == NTD_ID, Mode == mode)) |>
  inner_join(EIA_SEP_REPORT |> select(abbreviation, CO2_MWh),
             join_by(state == abbreviation))
```

```{r}
# EIA CO2 conversion factors (pounds of CO2 per gallon), from
# https://www.eia.gov/environment/emissions/co2_vol_mass.php
CO2_DIESEL   <- 22.45   # also used for bio-diesel
CO2_RESIDUAL <- 24.78   # bunker fuel
CO2_GASOLINE <- 18.73   # finished motor gasoline, also used for ethanol
CO2_KEROSENE <- 21.78
CO2_PROPANE  <- 12.68   # liquified petroleum gas
CO2_CNG      <- 120.85 * 0.12667   # natural gas per Mcf x Mcf per gallon equivalent
CO2_LNG      <- 120.85 * 0.0826    # natural gas per Mcf x Mcf per gallon of LNG

emissions <- emissions |>
  mutate(
    total_co2_lbs =
      (`Gallons of Diesel Fuel` + `Gallons of Bio-Diesel`) * CO2_DIESEL +
      `Gallons of Bunker Fuel`                             * CO2_RESIDUAL +
      (`Gallons of Gasoline` + `Gallons of Ethanol`)       * CO2_GASOLINE +
      `Gallons of Kerosene`                                * CO2_KEROSENE +
      `Gallons of Liquified Petroleum Gas`                 * CO2_PROPANE +
      `Gallons of Compressed Natural Gas`                  * CO2_CNG +
      `Gallons of Liquified Nat Gas`                       * CO2_LNG +
      # electricity: kWh -> MWh (divide by 1000) x the state's lbs CO2 per MWh
      (`Kilowatt hours of Electric Battery` +
         `Kilowatt hours of Electric Propulsion`) / 1000 * CO2_MWh,
    total_co2_tons = total_co2_lbs / 2000   # 2,000 lbs = 1 short ton
  )
```

I used inner_join to merge NTD_ENERGY with service_by_mode and then I merged that with EIA_SEP_REPORT. I joined them using inner_join because I needed ridership and fuel to compute the emissions. The Emissions were computed by multiplying each fuel with its EIA CO2 fuel factors data found in the EIA link [EIA's CO2 emissions coefficients](https://www.eia.gov/environment/emissions/co2_vol_mass.php).

Then the Electricity Emissions was computed by converting kWh to MWh (1MWh=1,000kWh) and then multiplying the MWh from NTD_ENERGY by the states CO2 per MWh that could be found in the EIA_SEP_REPORT. The fuel and electricity emissions are added for the total.

#### Normalize Emissions to Transit Usage (Emissions per trip and per mile)

```{r}
# Combine each agency's modes, then compute emissions per trip and per passenger mile.
# Very small agencies (under 100,000 trips a year) are dropped because their ratios
# swing wildly, and the rest are grouped by size so they compete with similar agencies.
agency_emissions <- emissions |>
  group_by(`NTD ID`, `Agency Name`, state) |>
  summarize(
    total_co2_lbs   = sum(total_co2_lbs),
    total_upt       = sum(total_upt),
    passenger_miles = sum(passenger_miles),
    .groups = "drop"
  ) |>
  filter(total_upt >= 100000, passenger_miles > 0) |>
  mutate(
    co2_per_trip = total_co2_lbs / total_upt,          # lbs of CO2 per trip
    co2_per_mile = total_co2_lbs / passenger_miles,    # lbs of CO2 per passenger mile
    size = case_when(
      total_upt < 1e6 ~ "Small",    # 100k to 1 million trips a year
      total_upt < 2e7 ~ "Medium",   # 1 to 20 million
      TRUE            ~ "Large"     # 20 million or more
    )
  )

# Most efficient agency in each size group (lowest CO2 per passenger mile)
agency_emissions |>
  group_by(size) |>
  slice_min(co2_per_mile, n = 3) |>
  ungroup() |>
  mutate(size = factor(size, levels = c("Small", "Medium", "Large"))) |>
  arrange(size, co2_per_mile) |>
  select(size, `Agency Name`, state, total_upt, co2_per_trip, co2_per_mile) |>
  gt(groupname_col = "size") |>
  tab_header(
    title = "Most Efficient Transit Agencies by Size",
    subtitle = "Lowest pounds of CO2 per passenger mile, 2025"
  ) |>
  cols_label(
    `Agency Name` = "Agency",
    state = "State",
    total_upt = "Trips",
    co2_per_trip = "CO2 per Trip (lbs)",
    co2_per_mile = "CO2 per Mile (lbs)"
  ) |>
  fmt_number(total_upt, decimals = 0) |>
  fmt_number(c(co2_per_trip, co2_per_mile), decimals = 3) |>
  tab_source_note(md("Sources: National Transit Database (2025); EIA State Electricity Profiles; EIA CO2 coefficients."))

# Greenest agency in each size group, for the awards
greenest <- agency_emissions |>
  group_by(size) |>
  slice_min(co2_per_mile, n = 1) |>
  ungroup()
# Greenest agency overall (lowest CO2 per passenger mile across all sizes)
overall_greenest <- agency_emissions |> slice_min(co2_per_mile, n = 1)
```

In the table above we can see the breakdown of the most efficient transit agencies by size after adding up each agency's modes (small-medium-large). We can also see the state they are in, the CO2 per trip, and the CO2 per mile. If the agency had 100k to 1 million trips it was categorized as small, if it had 1 million to 20 million trips it was categorized as medium, and if it had 20 million trips and above it was categorized large. The small agencies under 100k trips were not included in the analysis because the data is not reliable because it had wide variations and contained very few trips.

The data was grouped by size and divided by trips, and miles to be able to evaluate the efficiency of the transit agencies regardless of the size. Measuring the emissions per mile shows the efficiency of each agency and prevents the large agencies from being incorrectly labeled the least green just because of the volume they have, but show the efficiency of agencies in other size ranges.
